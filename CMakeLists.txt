cmake_minimum_required(VERSION 3.10)
project(QuantumTrade CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Threads REQUIRED)
include_directories(${CMAKE_SOURCE_DIR})
add_executable(quantumtrade backend/main.cpp)
target_link_libraries(quantumtrade Threads::Threads)
add_executable(test_engine tests/test_engine.cpp)
target_link_libraries(test_engine Threads::Threads)
if(WIN32)
  target_link_libraries(quantumtrade ws2_32)
endif()
enable_testing()
add_test(NAME engine_unit_tests COMMAND test_engine)
